# Runtime dependencies, pinned exactly so a rebuild installs what was tested.
#
# Previously only demoparser2 was pinned and everything else floated, which
# meant `docker compose up --build` resolved a different pandas/fastapi on every
# rebuild. These versions are the ones the test suite runs green against on
# Python 3.11 and 3.14.
#
# Only direct dependencies are listed. Transitive ones (starlette, uvloop,
# httptools, ...) are deliberately left to the resolver: uvicorn[standard] pulls
# a different set on Linux than on Windows, so a flat `pip freeze` taken on one
# platform produces a file that is wrong on the other.
#
# Dev tooling (pytest, ruff, mypy) lives in pyproject.toml: pip install -e ".[dev]"

pandas==3.0.2
fastapi==0.136.0
uvicorn[standard]==0.45.0
